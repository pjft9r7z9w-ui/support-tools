Create your Google Sheet

02
Add Apps Script webhook
In your Sheet: Extensions → Apps Script. Paste this code and click Deploy → Web app.
function doPost(e){ try{ var data=JSON.parse(e.postData.contents); var ss=SpreadsheetApp.getActiveSpreadsheet(); var type=data.type||"scorecard"; var sheet; if(type==="scorecard"){ sheet=ss.getSheetByName("QA Scorecard Data"); if(!sheet){sheet=ss.insertSheet("QA Scorecard Data"); sheet.appendRow(["Date","Agent","Evaluator","Channel","Score A","Score B","Final Score","Verdict","Strength","Action"]);} sheet.appendRow([data.date,data.agent,data.evaluator,data.channel,data.scoreA,data.scoreB,data.final,data.verdict,data.strength,data.action]); }else if(type==="pre_coaching_brief"){ sheet=ss.getSheetByName("Pre-Coaching Briefs"); if(!sheet){sheet=ss.insertSheet("Pre-Coaching Briefs"); sheet.appendRow(["Date","Agent","Manager","QA Score","Focus","Observations","Strengths","Q1","Q2","Q3","Q4"]);} sheet.appendRow([data.date,data.agent,data.manager,data.score,data.focus,data.observations,data.strengths,data.r1,data.r2,data.r3,data.r4]); }else if(type==="coaching_session"){ sheet=ss.getSheetByName("Coaching Sessions"); if(!sheet){sheet=ss.insertSheet("Coaching Sessions"); sheet.appendRow(["Date","Agent","Manager","Open✔","Explore✔","Develop✔","Close✔","Situation","Behavior","Impact","Root Cause","Solution","Agreements","Quiz Topic","Follow-up","Resources Sent"]);} sheet.appendRow([data.date,data.agent,data.manager,data.open_done,data.explore_done,data.develop_done,data.close_done,data.sbi_situation,data.sbi_behavior,data.sbi_impact,data.root_cause,data.solution,data.agreements,data.quiz_topic,data.followup_date,data.resources_sent]); }else if(type==="post_coaching_manager"){ sheet=ss.getSheetByName("Post-Coaching Resources"); if(!sheet){sheet=ss.insertSheet("Post-Coaching Resources"); sheet.appendRow(["Date","Agent","Manager","Session Focus","Skills","Quiz Deadline"]);} sheet.appendRow([data.date,data.agent,data.manager,data.session_focus,data.skills,data.quiz_deadline]); }else if(type==="post_coaching_agent"){ sheet=ss.getSheetByName("Post-Coaching Reflections"); if(!sheet){sheet=ss.insertSheet("Post-Coaching Reflections"); sheet.appendRow(["Date","Agent","Manager","Session Focus","Quiz Deadline","Q1 Most useful","Q2 Apply where","Q3 Still uncertain"]);} sheet.appendRow([data.date,data.agent,data.manager,data.session_focus,data.quiz_deadline,data.pq1,data.pq2,data.pq3]); } return ContentService.createTextOutput(JSON.stringify({status:"ok"})).setMimeType(ContentService.MimeType.JSON); }catch(err){ return ContentService.createTextOutput(JSON.stringify({status:"error",message:err.toString()})).setMimeType(ContentService.MimeType.JSON); } } function doGet(e){ return ContentService.createTextOutput(JSON.stringify({status:"ok",message:"SupportCraftStudio webhook active"})).setMimeType(ContentService.MimeType.JSON); }
